----------------------------------
_extract_capability_gap():
BEFORE: return finding.title or finding.category or "unknown_capability_gap"
AFTER: return finding.title or "" (empty if insufficient)

_extract_current_state():
BEFORE: return f"Cannot {parts[1].strip()}" if "cannot" in summary else "unable_to_perform_action"
AFTER: return f"Cannot {parts[1].strip()}" if "cannot" in summary else "" (empty if insufficient)

_extract_desired_state():
BEFORE: return finding.recommendation or "able_to_perform_action"
AFTER: return finding.recommendation or "" (empty if insufficient)

_extract_knowledge_required():
BEFORE: return f"knowledge_in_{finding.category}" if category else "general_knowledge"
AFTER: return finding.category if category in CAPABILITY_SHAPED_CATEGORIES else "" (empty if insufficient)
